#include <iostream>
using namespace std;

int main() {
    int arr[100], n, i, key;
    int low, high, mid, found = 0;

    cout << "Enter number of elements: ";
    cin >> n;

    cout << "Enter " << n << " elements in sorted order:\n";
    for (i = 0; i < n; i++) {
        cin >> arr[i];
    }

    cout << "Enter element to search: ";
    cin >> key;

    low = 0;
    high = n - 1;

    while (low <= high) {
        mid = (low + high) / 2;

        if (arr[mid] == key) {
            cout << "Element found at position " << mid + 1;
            found = 1;
            break;
        }
        else if (arr[mid] < key) {
            low = mid + 1;
        }
        else {
            high = mid - 1;
        }
    }

    if (!found) {
        cout << "Element not found";
    }

    return 0;
}
